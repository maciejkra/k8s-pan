Multi-stage build — Go binary do scratch

Single-stage (~900 MB)

FROM golang:1.22

WORKDIR /app

COPY . .

RUN go build -o app

# cały toolchain w obrazie

CMD ["./app"]

Multi-stage (~10 MB)

FROM golang:1.22 AS build

WORKDIR /app

COPY . .

RUN CGO_ENABLED=0 go build -o app

 

FROM scratch

COPY --from=build /app/app /

CMD ["/app"]
